% ── Figure: worked example (a) + SGPN sequential pipeline (b) ──────────────
\resizebox{\columnwidth}{!}{%
\begin{tikzpicture}[
    >=Stealth, font=\small,
    bus/.style    = {draw, rounded corners=3pt, align=center, font=\scriptsize},
    net/.style    = {circle, draw, minimum size=5.4mm, inner sep=0pt,
                     font=\scriptsize},
    tr/.style     = {thick},
    ontr/.style   = {line width=1.5pt, green!55!black},
    term/.style   = {font=\scriptsize},
    head/.style   = {draw, rounded corners=3pt, minimum width=1.35cm,
                     minimum height=0.72cm, align=center, fill=blue!8,
                     font=\scriptsize},
    mask/.style   = {draw, rounded corners=2pt, minimum width=1.32cm,
                     minimum height=0.30cm, align=center, fill=orange!20,
                     font=\tiny},
    arr/.style    = {->, thick},
    dep/.style    = {->, very thick, red!70!black},
    lbl/.style    = {font=\scriptsize, text=gray},
    pl/.style     = {font=\scriptsize\bfseries},
]
\ctikzset{transistors/scale=0.55}

%% ======================= Panel (a): worked example =======================
\node[pl] (pa) at (-4.55, 1.15) {(a)};

%% --- left: NMOS switch network for g = x1 x2 + x3 ---
% terminal rails
\draw[tr] (-4.25,-0.20) -- (-4.25,0.60);  \node[term] at (-4.45,0.48) {$S$};
\draw[tr] (-0.95,-0.20) -- (-0.95,0.60);  \node[term] at (-0.75,0.48) {$T$};
% top branch: S -- t1(x1) -- a -- t2(x2) -- T
\node[nmos, rotate=-90] (t1) at (-3.55,0.45) {};
\node[net, fill=white, minimum size=3.6mm] (ga) at (-2.60,0.45) {\scriptsize$a$};
\node[nmos, rotate=-90] (t2) at (-1.70,0.45) {};
\draw[tr] (-4.25,0.45) -- (t1.source);
\draw[tr] (t1.drain) -- (ga);
\draw[tr] (ga) -- (t2.source);
\draw[tr] (t2.drain) -- (-0.95,0.45);
\node[term, above=1pt] at (t1.gate) {$x_1$};
\node[term, above=1pt] at (t2.gate) {$x_2$};
% bottom branch: S -- t3(x3) -- T  (gate points down, label below)
\node[nmos, rotate=90] (t3) at (-2.60,-0.15) {};
\draw[tr] (-4.25,-0.15) -- (t3.drain);
\draw[tr] (t3.source) -- (-0.95,-0.15);
\node[term, below=1pt] at (t3.gate) {$x_3$};

%% --- middle: "modeled as" arrow ---
\draw[arr, line width=1pt] (-0.55,0.20) -- (0.35,0.20);
\node[lbl, align=center] at (-0.10,0.52) {modeled\\as};

%% --- right: switching graph (same network) ---
\node[net] (src) at (0.85,0.45) {$S$};
\node[net] (nb)  at (2.05,0.45) {$a$};
\node[net] (snk) at (3.25,0.45) {$T$};
\draw[ontr] (src) -- (nb) node[midway, above, font=\scriptsize, text=black] {$x_1$};
\draw[ontr] (nb)  -- (snk) node[midway, above, font=\scriptsize, text=black] {$x_2$};
\draw[tr] (src) to[bend right=26]
    node[midway, below=0pt, font=\scriptsize] {$x_3$} (snk);
\node[lbl, align=center] at (0.05,-0.42) {each transistor $\leftrightarrow$ one edge};

% divider between panels
\draw[gray!45, dashed] (-4.75, -0.64) -- (4.35, -0.64);

%% ================= Panel (b): SGPN sequential synthesis ==================
\node[pl] (pb) at (-4.55, -1.02) {(b)};

\node[bus, fill=green!8, minimum width=7.9cm] (ug) at (0.05, -1.32)
    {Unified graph $H_t = G_t \cup C_g$\; (partial network $+$ SOP target)};
\node[bus, fill=gray!10, minimum width=7.9cm, below=0.26cm of ug] (mpnn)
    {\textbf{MPNN} backbone ($L$ layers)\;$\Rightarrow$\;node emb.\ $\mathbf{h}_i$,\;
     summary $\mathbf{g}$};
\draw[arr] (ug) -- (mpnn);

%% four heads, drawn as a left-to-right dependent chain
\node[head, below=0.62cm of mpnn, xshift=-2.95cm] (h1) {Head 1\\source $u$};
\node[head, right=0.30cm of h1] (h2) {Head 2\\drain $v\!\mid\!u$};
\node[head, right=0.30cm of h2] (h3) {Head 3\\var.\ $k\!\mid\!u,v$};
\node[head, right=0.30cm of h3] (h4) {Head 4\\sign $s\!\mid\!u,v,k$};

% MPNN feeds every head
\foreach \h in {h1,h2,h3,h4} { \draw[arr] (mpnn.south) -- (\h.north); }
% sequential dependency: each selection conditions the next head
\draw[dep] (h1) -- (h2);
\draw[dep] (h2) -- (h3);
\draw[dep] (h3) -- (h4);

% per-head safety masks
\node[mask, below=0.16cm of h1] (m1) {\textit{mask}};
\node[mask, below=0.16cm of h2] (m2) {\textit{mask}};
\node[mask, below=0.16cm of h3] (m3) {\textit{mask}};
\node[mask, below=0.16cm of h4] (m4) {\textit{mask}};
\foreach \h/\m in {h1/m1,h2/m2,h3/m3,h4/m4} { \draw[arr] (\h)--(\m); }

% action + feedback loop
\node[bus, fill=red!7, minimum width=6.6cm, below=0.42cm of m2, xshift=1.02cm]
    (act) {Place transistor $(u^*,v^*,k^*,s^*)$ into $G_t$};
\foreach \m in {m1,m2,m3,m4} { \draw[arr] (\m.south) -- ++(0,-0.10) -| (act.north); }
% loop back to the unified graph (next step)
\draw[arr, blue!55!black] (act.east) -- ++(0.55,0)
    |- node[pos=0.25, right, font=\tiny, text=blue!55!black, align=left]
       {\;append,\\\;repeat} (ug.east);

\end{tikzpicture}%
}
